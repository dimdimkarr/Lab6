## Условие задачи
Задача 1. Напишите программу, которая проверяет, является ли введённый год високосным.
## Реализация программы
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

int check_year(int year);

int main(void)
{
    int year;
    int result;

    setlocale(LC_ALL, "Russian");

    printf("ЗАДАЧА 1: Проверка високосного года\n");
    printf("Введите год: ");

    if (scanf("%d", &year) != 1 || year <= 0)
    {
        printf("Ошибка: необходимо ввести положительное целое число.\n");
        return 1;
    }

    result = check_year(year);

    if (result)
        printf("Год %d високосный.\n\n", year);
    else
        printf("Год %d не високосный.\n\n", year);

    return 0;
}

int check_year(int year)
{
    int remainder;

    remainder = year % 4;

    if (remainder == 0)
    {
        if (year % 100 != 0 || year % 400 == 0)
            return 1;
        else
            return 0;
    }
    else
    {
        return 0;
    }
}
```

## Условие задачи
Задача 2. Напишите программу для вычисления значения функции F(x).
Вариант 6
## Контрольные примеры
* x = 0 → F(x) = 3 · 0 − 9 = −9
* x = 5 → F(x) = 3 · 5 − 9 = 6
* x = 7 → F(x) = 3 · 7 − 9 = 12
* x = 8 → F(x) = 1 / (64 − 4) = 1 / 60 ≈ 0,016667
* x = 10 → F(x) = 1 / (100 − 4) = 1 / 96 ≈ 0,010417
## Реализация программы
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

double f(double x);

int main(void)
{
    double x;
    double f_x;

    setlocale(LC_ALL, "Russian");

    printf("ЗАДАЧА 2: Вычисление функции F(x)\n");
    printf("Введите значение x: ");

    if (scanf("%lf", &x) != 1)
    {
        printf("Ошибка: необходимо ввести число.\n");
        return 1;
    }

    f_x = f(x);

    printf("Результат: F(%.2lf) = %.6lf\n", x, f_x);
    printf("Ветвь функции: %s\n",
        (x <= 7) ? "3x - 9" : "1/(x^2 - 4)");

    return 0;
}

/*
 * Контрольные примеры:
 * Ветвь 1 (x <= 7): F(x) = 3x - 9
 * x = 0  -> 3 * 0 - 9 = -9
 * x = 5  -> 3 * 5 - 9 = 6
 * x = 7  -> 3 * 7 - 9 = 12
 *
 * Ветвь 2 (x > 7): F(x) = 1 / (x^2 - 4)
 * x = 8  -> 1 / (64 - 4) = 1 / 60 = 0.016667
 * x = 10 -> 1 / (100 - 4) = 1 / 96 = 0.010417
 */

double f(double x)
{
    double y;

    if (x <= 7)
        y = 3.0 * x - 9.0;
    else
        y = 1.0 / (x * x - 4.0);

    return y;
}
```

# Домашнее задание. Вариант 7
## Условие задачи
Дано число N (0 ≤ N < 1000). Написать программу, которая проверяет, является ли сумма его цифр двузначным числом.
## Контрольные примеры
* N = 123 → 1 + 2 + 3 = 6 — сумма не двузначная
* N = 456 → 4 + 5 + 6 = 15 — сумма двузначная
* N = 999 → 9 + 9 + 9 = 27 — сумма двузначная
* N = 100 → 1 + 0 + 0 = 1 — сумма не двузначная
* N = 0 → сумма равна 0 — сумма не двузначная
## Реализация программы
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

int sum_digits(int n);

int main(void)
{
    int n;
    int sum;

    setlocale(LC_ALL, "Russian");

    printf("ДОМАШНЕЕ ЗАДАНИЕ. Вариант 7\n");
    printf("Введите число N (от 0 до 999): ");

    if (scanf("%d", &n) != 1 || n < 0 || n >= 1000)
    {
        printf("Ошибка: необходимо ввести число от 0 до 999.\n");
        return 1;
    }

    sum = sum_digits(n);

    printf("Сумма цифр числа %d: %d\n", n, sum);

    if (sum >= 10 && sum <= 99)
        printf("Сумма цифр является двузначным числом.\n");
    else
        printf("Сумма цифр не является двузначным числом.\n");

    return 0;
}

int sum_digits(int n)
{
    int hundreds;
    int tens;
    int units;
    int sum;

    hundreds = n / 100;
    tens = (n / 10) % 10;
    units = n % 10;

    sum = hundreds + tens + units;

    return sum;
}
```
